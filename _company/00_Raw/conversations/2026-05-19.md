/**
 * Price Sensitivity Mapper
 * 가격 민감도 맵을 기반으로 색상 비율 및 레이아웃 비율을 계산하는 알고리즘 모듈.
 */

interface MapParameters {
  R: number; // 좌표축 R (예: 가격 민감도 레벨)
  S: number; // 좌표축 S (예: 경쟁 환경/신뢰도 레벨)
}

interface MappingRules {
  colorMap: { [key: string]: string }; // R, S에 따른 색상 매핑 규칙
  aspectRatioMap: { [key: string]: number }; // R, S에 따른 Aspect Ratio 조정 규칙
}

/**
 * 가격 민감도 맵을 기반으로 동적 시각화 변수를 계산합니다.
 */
export class PriceSensitivityMapper {
  private rules: MappingRules;

  constructor(rules: MappingRules) {
    this.rules = rules;
    // 초기 검증 (필요시)
    if (!this.rules.colorMap || !this.rules.aspectRatioMap) {
      throw new Error("MappingRules에 colorMap 또는 aspectRatioMap이 누락되었습니다.");
    }
  }

  /**
   * 사용자 입력 (R, S)을 받아 동적 시각화 변수를 계산합니다.
   * @param params - 입력 좌표 (R, S)
   * @returns 계산된 시각화 변수 객체
   */
  public calculateVisualization(params: MapParameters): { colorRatio: Record<string, number>; aspectRatio: number } {
    const R = params.R;
    const S = params.S;

    // 1. 색상 비율 계산 (예시 로직)
    // R과 S를 기반으로 두 가지 주요 색상 채널(예: 레드 강조, 배경)의 비율을 결정
    let redRatio = this.rules.colorMap.red || 0.5; // 기본값 설정
    let blueRatio = this.rules.colorMap.blue || 0.5;

    // R이 높으면 (민감도가 높으면) 강조 색상 증가
    redRatio = Math.min(1, redRatio + R * 0.3);
    blueRatio = Math.max(0, blueRatio - S * 0.2);

    const colorRatio: Record<string, number> = {
      red: parseFloat(redRatio.toFixed(3)),
      blue: parseFloat(blueRatio.toFixed(3)),
      // 필요에 따라 다른 색상도 추가 가능 (예: green, yellow)
    };


    // 2. 레이아웃 블록 크기 조정 (Aspect Ratio 조정)
    // R과 S를 기반으로 컨테이너의 가로세로 비율을 결정하여 반응형 조정
    const aspectRatio = this.rules.aspectRatioMap[`R${S}`] || this.rules.aspectRatioMap[`${R}${S}`] || 16 / 9;

    return { colorRatio, aspectRatio };
  }
}

import { PriceSensitivityMapper } from '../src/price_sensitivity_mapper';

describe('PriceSensitivityMapper', () => {
  // Designer가 제시한 규칙을 임시로 정의 (실제 값은 designer.md에서 가져와야 함)

    },

      'R3S2': 4 / 3, // R=3, S=2 일 때 4:3 비율
    }

  let mapper: PriceSensitivityMapper;

  beforeEach(() => {
    // 각 테스트마다 규칙을 초기화하여 독립성을 보장
    mapper = new PriceSensitivityMapper(mockRules);
  });

  test('should correctly calculate visualization for a standard input (R=1, S=1)', () => {
    const params = { R: 1, S: 1 };
    const result = mapper.calculateVisualization(params);

    // R=1, S=1 입력 시, colorRatio는 기본값에서 약간 조정되어야 함. Aspect Ratio는 'R1S1' 규칙 적용.
    expect(result.aspectRatio).toBe(1 / 1);
    expect(result.colorRatio.red).toBeGreaterThanOrEqual(0.1); // 최소한의 레드 비율은 유지
  });

  test('should apply complex rules for a specific input (R=3, S=2)', () => {
    const params = { R: 3, S: 2 };
    const result = mapper.calculateVisualization(params);

    // R=3, S=2 입력 시, aspectRatio는 'R3S2' 규칙 적용
    expect(result.aspectRatio).toBe(4 / 3);
  });

  test('should throw error if rules are missing', () => {
    const incompleteRules = { colorMap: { red: 0.1 } }; // aspectRatioMap 누락
    expect(() => new PriceSensitivityMapper(incompleteRules)).toThrow("MappingRules에 colorMap 또는 aspectRatioMap이 누락되었습니다.");
  });
});

## 1. 목표 및 입력 정의
**목표:** T2A CR 신뢰도 변화 시나리오(High, Medium, Low)에 따른 실제 썸네일 A/B 테스트 결과를 예측하고 최적 가격을 산출하는 로직 개발 착수 준비.
**핵심 입력 변수:**
1.  $T_{CR}$: 현재 T2A CR 신뢰도 (High, Medium, Low 중 하나)
2.  $R$: 가격 민감도 맵의 수평 좌표축 값 (0~100 범위 가정)
3.  $S$: 가격 민감도 맵의 수직 좌표축 값 (0~100 범위 가정)
4.  $\text{Baseline Data}$: 기준이 되는 영상/릴스 초기 반응 데이터 (클릭률, 전환율 등)

## 2. T2A CR 시나리오별 가중치 매핑 (Designer Input)
T2A CR의 신뢰도 변화에 따라 가격 민감도를 해석하는 가중치를 정의합니다. 이 값들은 $R$과 $S$를 변환하여 최종 위험 점수 및 승률 계산에 사용됩니다.

| T2A CR 시나리오 | 가격 민감도 해석 (Mapping Rule) | 위험 감수 수준 ($W_{Risk}$) | 최적 가격 조정 계수 ($\alpha$) |
| :--- | :--- | :--- | :--- |
| **High** | 높은 신뢰도. 안정적인 수익 추구. | $W_{Risk} = 0.2$ (낮음) | $\alpha = 1.0$ (기준 가격 유지) |
| **Medium** | 중간 신뢰도. 균형 잡힌 접근. | $W_{Risk} = 0.5$ (중간) | $\alpha = 1.2$ (약간의 공격적 조정) |
| **Low** | 낮은 신뢰도. 리스크 회피 중시. | $W_{Risk} = 0.8$ (높음) | $\alpha = 1.5$ (보수적 가격 책정) |

## 3. A/B 테스트 결과 산출 로직 (Algorithm Blueprint)
최종 A/B 테스트 결과($\text{Result}$)는 다음 공식을 기반으로 계산됩니다.

$$\text{Risk Score} = (\text{Baseline Loss Rate}) \times W_{Risk} + (R \times S \times \alpha)$$

1.  **$\text{Loss Rate}$ 산출:** $\text{Loss Rate} = f(\text{Baseline Data}, T_{CR})$
    *   $T_{CR}=\text{High}$: $\text{Loss Rate} = \text{Baseline Loss Rate} \times 0.5$
    *   $T_{CR}=\text{Medium}$: $\text{Loss Rate} = \text{Baseline Loss Rate}$
    *   $T_{CR}=\text{Low}$: $\text{Loss Rate} = \text{Baseline Loss Rate} \times 1.5$

2.  **최적 가격 ($\mathbf{P_{opt}}$) 산출:** $P_{opt}$는 위험 점수($\text{Risk Score}$)를 최소화하는 지점을 찾습니다.
    $$P_{opt} = P_{current} - (\beta \times \text{Risk Score})$$
    *   $\beta$: 가격 민감도에 따른 민감도 상수 (Designer가 최종 설정)

3.  **A/B 테스트 승률 예측:** $T_{CR}$ 상태에 따라 $\text{Loss Rate}$를 보정하여 최종 예상 전환율을 산출합니다.
    $$\text{Predicted Conversion} = \text{Baseline Conversion} - (\text{Risk Score} \times \gamma)$$
    *   $\gamma$: 위험 감수 계수 ($\gamma$는 $W_{Risk}$와 연동)

def calculate_ab_test_results(r: float, s: float, t2a_cr_reliability: float) -> dict:
    """
    가격 민감도 맵 좌표와 T2A CR 신뢰도를 기반으로 A/B 테스트 결과를 계산합니다.

    Args:
        r (float): 가격 민감도 축 R 값 (0 ~ 1).
        s (float): 수익성 민감도 축 S 값 (0 ~ 1).
        t2a_cr_reliability (float): T2A CR 신뢰도 (0.0 ~ 1.0).

    Returns:
        dict: 계산된 결과 및 시각화에 필요한 변수 포함.
    """
    if not (0 <= r <= 1 and 0 <= s <= 1 and 0 <= t2a_cr_reliability <= 1):
        raise ValueError("Input parameters R, S, T2A CR Reliability must be between 0 and 1.")

    # Designer에서 정의된 색상 매핑 규칙 (예시: Red 강조)
    # 신뢰도가 낮을수록(T2A CR Reliability가 낮을수록), 위험 점수와 민감도는 증가한다고 가정.
    risk_score = r * (1 - t2a_cr_reliability) + s * (1 - t2a_cr_reliability)
    
    # A/B 테스트 결과 시뮬레이션 (실제 비즈니스 로직에 따라 조정 필요)
    # 신뢰도가 낮을수록, 민감도 영역 내에서 변동성이 커진다고 가정하여 위험도를 증폭.
    ab_test_outcome = {
        "risk_level": round(risk_score * 100, 2), # 백분율로 표현
        "sensitivity_metric": round(r + s, 2), # 단순 민감도 합계
        "confidence_margin": round((1 - t2a_cr_reliability) * 100, 2), # 신뢰도 손실 마진
        "suggested_action": "Review Pricing Model" if risk_score > 0.5 else "Monitor"
    }

    return ab_test_outcome

$$ \text{RiskScore} = R \cdot (1 - T2A\ CR) + S \cdot (1 - T2A\ CR) $$

$$ \text{ColorIntensity} = 1.0 - T2A\ CR $$
*   **설명:** 시각적 강조(레드)의 강도를 시스템 신뢰도에 반비례하게 설정하여, 데이터 불확실성이 높을수록 시각적으로 더 강한 경고를 제공합니다.

# 시스템 구조 설명: 가격 민감도 시각화 프레임워크 (Blueprint & Manual)

## 1. 아키텍처 컴포넌트 역할
이 시스템은 데이터 파이프라인의 신뢰도를 기반으로 가격 책정 전략의 위험도를 정량화하여 시각화하는 것을 목표로 합니다.

*   **Data Ingestion Layer:** 외부 데이터(T2A CR 신뢰도 등)를 수집하고 정제합니다. (API 연동 및 검증 담당)
*   **Core Calculation Engine (`price_sensitivity_calculator.py`):** 입력된 민감도 좌표($R, S$)와 시스템 신뢰도($T2A\ CR$)를 받아 A/B 테스트 결과 및 시각화 변수를 계산하는 핵심 알고리즘입니다. (수학적 모델 구현)
*   **API Integration Layer (`backend/main.py`):** Core Engine의 계산 결과를 외부 서비스(프론트엔드 대시보드)가 요청할 수 있도록 RESTful 엔드포인트로 제공합니다. (데이터 제공 인터페이스)

## 2. 핵심 로직: 사일런트 실패 감지 및 복구 패턴
핵심은 데이터 입력에 대한 **복원력(Resilience)** 확보입니다.

### A. 신뢰도 기반 위험 점수 산출
$$ \text{RiskScore} = R \cdot (1 - T2A\ CR) + S \cdot (1 - T2A\ CR) $$
*   **설명:** 가격 민감도 영역 내에서 시스템의 신뢰도가 낮을수록(즉, $1 - T2A\ CR$ 값이 클수록), 해당 지점에서의 위험 점수가 선형적으로 증가하도록 정의했습니다.

### B. 시각화 변수 매핑 규칙
$$ \text{ColorIntensity} = 1.0 - T2A\ CR $$
*   **설명:** 시각적 강조(레드)의 강도를 시스템 신뢰도에 반비례하게 설정하여, 데이터 불확실성이 높을수록 시각적으로 더 강한 경고를 제공합니다.

### C. 복구 로직 (Circuit Breaker 패턴 적용 준비)
API 호출이나 외부 데이터 연결 실패 시, **`backend/main.py`** 내에서 예외 처리를 통해 즉시 응답을 중단하고 시스템 상태를 기록해야 합니다. 이는 다운스트림 프로세스가 무한 대기하는 것을 방지합니다.
